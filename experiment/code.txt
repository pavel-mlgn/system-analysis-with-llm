package com.example.limit.config;

import org.apache.kafka.clients.admin.NewTopic;
import org.apache.kafka.common.TopicPartition;
import org.apache.kafka.common.serialization.ByteArraySerializer;
import org.apache.kafka.common.serialization.LongSerializer;
import org.apache.kafka.common.serialization.Serializer;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.kafka.config.TopicBuilder;
import org.springframework.kafka.core.DefaultKafkaProducerFactory;
import org.springframework.kafka.core.KafkaOperations;
import org.springframework.kafka.core.KafkaTemplate;
import org.springframework.kafka.core.ProducerFactory;
import org.springframework.kafka.listener.DeadLetterPublishingRecoverer;
import org.springframework.kafka.listener.DefaultErrorHandler;
import org.springframework.kafka.support.serializer.DelegatingByTypeSerializer;
import org.springframework.kafka.support.serializer.JsonSerializer;
import org.springframework.util.backoff.FixedBackOff;

import java.util.LinkedHashMap;
import java.util.Map;

@Configuration
public class KafkaErrorHandlingConfig {

    @Bean
    public NewTopic transfersDlt() {
        return TopicBuilder.name("transfers.DLT").partitions(3).build();
    }

    @Bean
    public DefaultErrorHandler kafkaErrorHandler(ProducerFactory<?, ?> producerFactory) {
        Map<String, Object> producerConfig = producerFactory.getConfigurationProperties();
        KafkaTemplate<Object, byte[]> rawTemplate = new KafkaTemplate<>(new DefaultKafkaProducerFactory<>(
                producerConfig, keySerializer(), new ByteArraySerializer()));
        KafkaTemplate<Object, Object> jsonTemplate = new KafkaTemplate<>(new DefaultKafkaProducerFactory<>(
                producerConfig, keySerializer(), new JsonSerializer<>()));

        Map<Class<?>, KafkaOperations<?, ?>> templates = new LinkedHashMap<>();
        templates.put(byte[].class, rawTemplate);
        templates.put(Object.class, jsonTemplate);

        DeadLetterPublishingRecoverer recoverer = new DeadLetterPublishingRecoverer(templates,
                (record, ex) -> new TopicPartition(record.topic() + ".DLT", record.partition()));
        recoverer.setFailIfSendResultIsError(false);
        return new DefaultErrorHandler(recoverer, new FixedBackOff(1000L, 3L));
    }

    private static Serializer<Object> keySerializer() {
        return new DelegatingByTypeSerializer(Map.of(
                byte[].class, new ByteArraySerializer(),
                Long.class, new LongSerializer()));
    }
}

import jakarta.persistence.*;
import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.LocalDateTime;

@Entity
@Table(name = "reservations")
public class Reservation {

    public static final String APPROVED = "APPROVED";
    public static final String DECLINED = "DECLINED";

    public static final String STATE_RESERVED = "RESERVED";
    public static final String STATE_COMMITTED = "COMMITTED";
    public static final String STATE_RELEASED = "RELEASED";
    public static final String STATE_NONE = "NONE";

    @Id
    private Long transferId;
    private Long senderId;
    private BigDecimal amount;
    private String decision;
    private String reason;
    private String state;
    private LocalDate reservedOn;
    private LocalDateTime createdAt;
    private LocalDateTime updatedAt;

    public Reservation() {
    }

    public Reservation(Long transferId, Long senderId, BigDecimal amount, LocalDate reservedOn) {
        this.transferId = transferId;
        this.senderId = senderId;
        this.amount = amount;
        this.reservedOn = reservedOn;
        this.createdAt = LocalDateTime.now();
        this.updatedAt = this.createdAt;
    }

    public void approve() {
        this.decision = APPROVED;
        this.reason = null;
        this.state = STATE_RESERVED;
    }

    public void decline(String reason) {
        this.decision = DECLINED;
        this.reason = reason;
        this.state = STATE_NONE;
    }

    public void markCommitted() {
        this.state = STATE_COMMITTED;
        this.updatedAt = LocalDateTime.now();
    }

    public void markReleased() {
        this.state = STATE_RELEASED;
        this.updatedAt = LocalDateTime.now();
    }

    public boolean isReserved() {
        return STATE_RESERVED.equals(state);
    }

    public Long getTransferId() { return transferId; }
    public void setTransferId(Long transferId) { this.transferId = transferId; }
    public Long getSenderId() { return senderId; }
    public void setSenderId(Long senderId) { this.senderId = senderId; }

package com.example.limit.domain;

import jakarta.persistence.*;
import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.YearMonth;

@Entity
@Table(name = "sender_usage")
public class SenderUsage {

    @Id
    private Long senderId;
    private BigDecimal dailyUsed;
    private BigDecimal monthlyUsed;
    private LocalDate lastDailyReset;
    private LocalDate lastMonthlyReset;

    @Version
    private Long version;

    public SenderUsage() {
    }

    public SenderUsage(Long senderId, LocalDate today) {
        this.senderId = senderId;
        this.dailyUsed = BigDecimal.ZERO;
        this.monthlyUsed = BigDecimal.ZERO;
        this.lastDailyReset = today;
        this.lastMonthlyReset = today;
    }

    public void lazyReset(LocalDate today) {
        if (!today.equals(lastDailyReset)) {
            dailyUsed = BigDecimal.ZERO;
            lastDailyReset = today;
        }
        if (!YearMonth.from(today).equals(YearMonth.from(lastMonthlyReset))) {
            monthlyUsed = BigDecimal.ZERO;
            lastMonthlyReset = today;
        }
    }

    public void reserve(BigDecimal amount) {
        dailyUsed = dailyUsed.add(amount);
        monthlyUsed = monthlyUsed.add(amount);
    }

    public void release(BigDecimal amount, LocalDate reservedOn, LocalDate today) {
        lazyReset(today);
        if (reservedOn.equals(today)) {
            dailyUsed = dailyUsed.subtract(amount).max(BigDecimal.ZERO);
        }
        if (YearMonth.from(reservedOn).equals(YearMonth.from(today))) {
            monthlyUsed = monthlyUsed.subtract(amount).max(BigDecimal.ZERO);
        }
    }

    public Long getSenderId() { return senderId; }
    public void setSenderId(Long senderId) { this.senderId = senderId; }
    public BigDecimal getDailyUsed() { return dailyUsed; }
    public void setDailyUsed(BigDecimal dailyUsed) { this.dailyUsed = dailyUsed; }
    public BigDecimal getMonthlyUsed() { return monthlyUsed; }
    public void setMonthlyUsed(BigDecimal monthlyUsed) { this.monthlyUsed = monthlyUsed; }
    public LocalDate getLastDailyReset() { return lastDailyReset; }
    public void setLastDailyReset(LocalDate lastDailyReset) { this.lastDailyReset = lastDailyReset; }
    public LocalDate getLastMonthlyReset() { return lastMonthlyReset; }
    public void setLastMonthlyReset(LocalDate lastMonthlyReset) { this.lastMonthlyReset = lastMonthlyReset; }
    public Long getVersion() { return version; }
    public void setVersion(Long version) { this.version = version; }

    @Override
    public String toString() {
        return "SenderUsage{" +
                "senderId=" + senderId +
                ", dailyUsed=" + dailyUsed +
                ", monthlyUsed=" + monthlyUsed +
                ", lastDailyReset=" + lastDailyReset +
                ", lastMonthlyReset=" + lastMonthlyReset +
                ", version=" + version +

package com.example.limit.service;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import com.example.common.Transfer;
import com.example.limit.domain.Reservation;
import com.example.limit.domain.SenderUsage;
import com.example.limit.repository.ReservationRepository;
import com.example.limit.repository.SenderUsageRepository;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.Optional;

@Service
public class LimitLedger {

    private static final Logger LOG = LoggerFactory.getLogger(LimitLedger.class);

    private final SenderUsageRepository usageRepository;
    private final ReservationRepository reservationRepository;

    @Value("${limits.daily-max}")
    private BigDecimal dailyMax;

    @Value("${limits.monthly-max}")
    private BigDecimal monthlyMax;

    public LimitLedger(SenderUsageRepository usageRepository, ReservationRepository reservationRepository) {
        this.usageRepository = usageRepository;
        this.reservationRepository = reservationRepository;
    }

    @Transactional
    public Reservation decide(Transfer transfer) {
        Optional<Reservation> existing = reservationRepository.findById(transfer.id());
        if (existing.isPresent()) {
            LOG.info("Duplicate NEW for transfer {}, returning stored decision", transfer.id());
            return existing.get();
        }

        LocalDate today = LocalDate.now();
        BigDecimal amount = transfer.amount();

        if (amount == null || amount.signum() <= 0) {
            Reservation invalid = new Reservation(transfer.id(), transfer.senderId(), BigDecimal.ZERO, today);
            invalid.decline("INVALID_AMOUNT");
            return reservationRepository.save(invalid);
        }

        Reservation reservation = new Reservation(transfer.id(), transfer.senderId(), amount, today);
        SenderUsage usage = usageRepository.findById(transfer.senderId())
                .orElseGet(() -> new SenderUsage(transfer.senderId(), today));
        usage.lazyReset(today);
        LOG.info("Sender {} usage before decision: daily={}, monthly={}",
                usage.getSenderId(), usage.getDailyUsed(), usage.getMonthlyUsed());

        if (usage.getDailyUsed().add(amount).compareTo(dailyMax) > 0) {
            reservation.decline("DAILY_LIMIT_EXCEEDED");
        } else if (usage.getMonthlyUsed().add(amount).compareTo(monthlyMax) > 0) {
            reservation.decline("MONTHLY_LIMIT_EXCEEDED");
        } else {
            usage.reserve(amount);
            reservation.approve();
        }

        usageRepository.save(usage);
        return reservationRepository.save(reservation);
    }

    @Transactional
    public void settle(Transfer transfer) {
        Optional<Reservation> found = reservationRepository.findById(transfer.id());
        if (found.isEmpty()) {
            LOG.warn("No reservation for transfer {}, final status {} skipped", transfer.id(), transfer.status());
            return;
        }
        Reservation reservation = found.get();
        if (!reservation.isReserved()) {
            LOG.info("Transfer {} is in state {}, nothing to settle", transfer.id(), reservation.getState());
            return;
        }

        switch (transfer.status()) {
            case "COMPLETED" -> reservation.markCommitted();
            case "CANCELLED", "DENIED" -> {
                SenderUsage usage = usageRepository.findById(reservation.getSenderId()).orElseThrow();
                usage.release(reservation.getAmount(), reservation.getReservedOn(), LocalDate.now());
                usageRepository.save(usage);
                reservation.markReleased();
            }
            default -> {
                LOG.warn("Unexpected final status {} for transfer {}", transfer.status(), transfer.id());
                return;
            }
        }
        reservationRepository.save(reservation);
        LOG.info("Reservation for transfer {} settled: {}", reservation.getTransferId(), reservation.getState());
    }
}

package com.example.limit.service;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.kafka.core.KafkaTemplate;
import org.springframework.stereotype.Service;
import com.example.common.Transfer;
import com.example.limit.domain.Reservation;

import java.util.concurrent.ExecutionException;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.TimeoutException;

@Service
public class LimitService {

    private static final String DECIDED_BY = "limit";
    private static final Logger LOG = LoggerFactory.getLogger(LimitService.class);
    private final LimitLedger ledger;
    private final KafkaTemplate<Long, Transfer> template;

    public LimitService(LimitLedger ledger, KafkaTemplate<Long, Transfer> template) {
        this.ledger = ledger;
        this.template = template;
    }

    public void onNewTransfer(Transfer transfer) {
        Reservation reservation = ledger.decide(transfer);
        sendAndWait(transfer.withDecision(reservation.getDecision(), DECIDED_BY, reservation.getReason()));
    }

    public void onFinalStatus(Transfer transfer) {
        ledger.settle(transfer);
    }

    private void sendAndWait(Transfer transfer) {
        try {
            template.send("limit-results", transfer.id(), transfer).get(10, TimeUnit.SECONDS);
            LOG.info("Published {} for transfer {}", transfer.status(), transfer.id());
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            throw new IllegalStateException("Interrupted while sending to limit-results", e);
        } catch (ExecutionException | TimeoutException e) {
            throw new IllegalStateException("Failed to send to limit-results", e);
        }
    }
}

import org.apache.kafka.clients.admin.NewTopic;
import org.apache.kafka.common.TopicPartition;
import org.apache.kafka.common.serialization.ByteArraySerializer;
import org.apache.kafka.common.serialization.LongSerializer;
import org.apache.kafka.common.serialization.Serializer;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.kafka.config.TopicBuilder;
import org.springframework.kafka.core.DefaultKafkaProducerFactory;
import org.springframework.kafka.core.KafkaOperations;
import org.springframework.kafka.core.KafkaTemplate;
import org.springframework.kafka.core.ProducerFactory;
import org.springframework.kafka.listener.DeadLetterPublishingRecoverer;
import org.springframework.kafka.listener.DefaultErrorHandler;
import org.springframework.kafka.support.serializer.DelegatingByTypeSerializer;
import org.springframework.kafka.support.serializer.JsonSerializer;
import org.springframework.util.backoff.FixedBackOff;

import java.util.LinkedHashMap;
import java.util.Map;

@Configuration
public class KafkaErrorHandlingConfig {

    @Bean
    public NewTopic transfersDlt() {
        return TopicBuilder.name("transfers.DLT").partitions(3).build();
    }

    @Bean
    public DefaultErrorHandler kafkaErrorHandler(ProducerFactory<?, ?> producerFactory) {
        Map<String, Object> producerConfig = producerFactory.getConfigurationProperties();
        KafkaTemplate<Object, byte[]> rawTemplate = new KafkaTemplate<>(new DefaultKafkaProducerFactory<>(
                producerConfig, keySerializer(), new ByteArraySerializer()));
        KafkaTemplate<Object, Object> jsonTemplate = new KafkaTemplate<>(new DefaultKafkaProducerFactory<>(
                producerConfig, keySerializer(), new JsonSerializer<>()));

        Map<Class<?>, KafkaOperations<?, ?>> templates = new LinkedHashMap<>();
        templates.put(byte[].class, rawTemplate);
        templates.put(Object.class, jsonTemplate);

        DeadLetterPublishingRecoverer recoverer = new DeadLetterPublishingRecoverer(templates,
                (record, ex) -> new TopicPartition(record.topic() + ".DLT", record.partition()));
        recoverer.setFailIfSendResultIsError(false);
        return new DefaultErrorHandler(recoverer, new FixedBackOff(1000L, 3L));
    }

    private static Serializer<Object> keySerializer() {
        return new DelegatingByTypeSerializer(Map.of(
                byte[].class, new ByteArraySerializer(),
                Long.class, new LongSerializer()));
    }
}

package com.example.screening.service;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import com.example.common.Transfer;
import com.example.screening.domain.ScreeningCheck;
import com.example.screening.repository.BlockedSenderRepository;
import com.example.screening.repository.ScreeningCheckRepository;

import java.util.Optional;

@Service
public class ScreeningJournal {

    private static final Logger LOG = LoggerFactory.getLogger(ScreeningJournal.class);

    private final BlockedSenderRepository blockedSenderRepository;
    private final ScreeningCheckRepository checkRepository;

    public ScreeningJournal(BlockedSenderRepository blockedSenderRepository,
                            ScreeningCheckRepository checkRepository) {
        this.blockedSenderRepository = blockedSenderRepository;
        this.checkRepository = checkRepository;
    }

    @Transactional
    public ScreeningCheck check(Transfer transfer) {
        Optional<ScreeningCheck> existing = checkRepository.findById(transfer.id());
        if (existing.isPresent()) {
            LOG.info("Duplicate NEW for transfer {}, returning stored check", transfer.id());
            return existing.get();
        }

        var blocked = blockedSenderRepository.findBySenderId(transfer.senderId());
        ScreeningCheck check;
        if (blocked.isPresent()) {
            LOG.info("Sender {} is blocked: {}", transfer.senderId(), blocked.get());
            check = new ScreeningCheck(transfer.id(), transfer.senderId(),
                    ScreeningCheck.DECLINED, "SENDER_BLOCKED");
        } else {
            check = new ScreeningCheck(transfer.id(), transfer.senderId(),
                    ScreeningCheck.APPROVED, null);
        }
        return checkRepository.save(check);
    }
}

package com.example.screening.service;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.kafka.core.KafkaTemplate;
import org.springframework.stereotype.Service;
import com.example.common.Transfer;
import com.example.screening.domain.ScreeningCheck;

import java.util.concurrent.ExecutionException;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.TimeoutException;

@Service
public class ScreeningService {

    private static final String DECIDED_BY = "screening";
    private static final Logger LOG = LoggerFactory.getLogger(ScreeningService.class);
    private final ScreeningJournal journal;
    private final KafkaTemplate<Long, Transfer> template;

    public ScreeningService(ScreeningJournal journal, KafkaTemplate<Long, Transfer> template) {
        this.journal = journal;
        this.template = template;
    }

    public void onNewTransfer(Transfer transfer) {
        ScreeningCheck check = journal.check(transfer);
        sendAndWait(transfer.withDecision(check.getResult(), DECIDED_BY, check.getReason()));
    }

    private void sendAndWait(Transfer transfer) {
        try {
            template.send("screening-results", transfer.id(), transfer).get(10, TimeUnit.SECONDS);
            LOG.info("Published {} for transfer {}", transfer.status(), transfer.id());
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            throw new IllegalStateException("Interrupted while sending to screening-results", e);
        } catch (ExecutionException | TimeoutException e) {
            throw new IllegalStateException("Failed to send to screening-results", e);
        }
    }
}

package com.example.transfer.config;

import org.apache.kafka.common.serialization.Serdes;
import org.apache.kafka.streams.StreamsBuilder;
import org.apache.kafka.streams.kstream.Consumed;
import org.apache.kafka.streams.kstream.JoinWindows;
import org.apache.kafka.streams.kstream.KStream;
import org.apache.kafka.streams.kstream.StreamJoined;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.kafka.annotation.EnableKafkaStreams;
import org.springframework.kafka.support.serializer.JsonSerde;
import com.example.common.Transfer;
import com.example.transfer.service.TransferDecisionService;

import java.time.Duration;

@Configuration
@EnableKafkaStreams
public class DecisionTopologyConfig {

    private static final Logger LOG = LoggerFactory.getLogger(DecisionTopologyConfig.class);

    @Bean
    public KStream<Long, Transfer> decisionStream(StreamsBuilder builder, TransferDecisionService decisionService) {
        JsonSerde<Transfer> transferSerde = new JsonSerde<>(Transfer.class);
        KStream<Long, Transfer> limitResults = builder
                .stream("limit-results", Consumed.with(Serdes.Long(), transferSerde));

        limitResults.join(
                        builder.stream("screening-results", Consumed.with(Serdes.Long(), transferSerde)),
                        decisionService::decide,
                        JoinWindows.ofTimeDifferenceWithNoGrace(Duration.ofSeconds(10)),
                        StreamJoined.with(Serdes.Long(), transferSerde, transferSerde))
                .peek((key, decision) -> LOG.info("Decision for transfer {}: {}", decision.id(), decision.status()))
                .to("transfers");

        return limitResults;
    }
}

                producerConfig, keySerializer(), new ByteArraySerializer()));
        KafkaTemplate<Object, Object> jsonTemplate = new KafkaTemplate<>(new DefaultKafkaProducerFactory<>(
                producerConfig, keySerializer(), new JsonSerializer<>()));

        Map<Class<?>, KafkaOperations<?, ?>> templates = new LinkedHashMap<>();
        templates.put(byte[].class, rawTemplate);
        templates.put(Object.class, jsonTemplate);

        DeadLetterPublishingRecoverer recoverer = new DeadLetterPublishingRecoverer(templates,
                (record, ex) -> new TopicPartition(record.topic() + ".DLT", record.partition()));
        recoverer.setFailIfSendResultIsError(false);
        return new DefaultErrorHandler(recoverer, new FixedBackOff(1000L, 3L));
    }

    private static Serializer<Object> keySerializer() {
        return new DelegatingByTypeSerializer(Map.of(
                byte[].class, new ByteArraySerializer(),
                Long.class, new LongSerializer()));

package com.example.transfer.controller;

import jakarta.validation.Valid;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.dao.DataIntegrityViolationException;
import org.springframework.http.HttpStatus;
import org.springframework.http.ProblemDetail;

import org.springframework.kafka.KafkaException;
import org.springframework.kafka.core.KafkaTemplate;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.server.ResponseStatusException;
import com.example.common.Transfer;
import com.example.transfer.domain.TransferEntity;
import com.example.transfer.domain.TransferHistoryEntry;
import com.example.transfer.repository.TransferHistoryRepository;
import com.example.transfer.repository.TransferRepository;
import com.example.transfer.service.TransferTrackingService;

import java.util.Comparator;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.concurrent.ExecutionException;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.TimeoutException;

@RestController
@RequestMapping("/transfers")
public class TransferController {

    private static final Logger LOG = LoggerFactory.getLogger(TransferController.class);
    private static final String STATUS_NEW = "NEW";
    private static final String KAFKA_UNAVAILABLE =
            "Kafka недоступна, перевод сохранён — повторите запрос с тем же id";

    private final KafkaTemplate<Long, Transfer> template;
    private final TransferRepository transferRepository;
    private final TransferHistoryRepository historyRepository;
    private final TransferTrackingService trackingService;

    public TransferController(KafkaTemplate<Long, Transfer> template,
                              TransferRepository transferRepository,
                              TransferHistoryRepository historyRepository,
                              TransferTrackingService trackingService) {
        this.template = template;
        this.transferRepository = transferRepository;
        this.historyRepository = historyRepository;
        this.trackingService = trackingService;
    }

    @PostMapping
    public ResponseEntity<?> create(@Valid @RequestBody TransferRequest request) {
        var existing = transferRepository.findById(request.id());
        if (existing.isPresent()) {
            return duplicate(existing.get(), request);
        }

        TransferEntity entity;
        try {
            entity = trackingService.createTransfer(request);
        } catch (DataIntegrityViolationException e) {
            return transferRepository.findById(request.id())
                    .map(found -> duplicate(found, request))
                    .orElseThrow(() -> e);
        }

        publishNew(entity);
        return ResponseEntity.status(HttpStatus.CREATED).body(entity);
    }

    @GetMapping
    public List<TransferEntity> all() {
        return transferRepository.findAllByOrderByCreatedAtDesc();
    }

    @GetMapping("/{id}")
    public ResponseEntity<Map<String, Object>> getById(@PathVariable Long id) {
        var transfer = transferRepository.findById(id);
        if (transfer.isEmpty()) {
            return ResponseEntity.notFound().build();
        }
        List<TransferHistoryEntry> history = historyRepository.findByTransferIdOrderByChangedAtAsc(id).stream()
                .sorted(Comparator.comparing(TransferHistoryEntry::getStage)
                        .thenComparing(TransferHistoryEntry::getChangedAt))
                .toList();

        Map<String, Object> response = new LinkedHashMap<>();
        response.put("transfer", transfer.get());
        response.put("history", history);
        return ResponseEntity.ok(response);
    }

    private ResponseEntity<?> duplicate(TransferEntity existing, TransferRequest request) {
        if (!sameParameters(existing, request)) {
            LOG.warn("Transfer id={} already exists with other parameters", request.id());
            ProblemDetail problem = ProblemDetail.forStatusAndDetail(HttpStatus.CONFLICT,
                    "Перевод с id=" + request.id() + " уже существует с другими параметрами");
            return ResponseEntity.status(HttpStatus.CONFLICT).body(problem);
        }
        if (STATUS_NEW.equals(existing.getStatus())) {
            publishNew(existing);
        }
        LOG.info("Duplicate transfer id={}, returning existing", request.id());
        return ResponseEntity.ok(existing);
    }

    private boolean sameParameters(TransferEntity existing, TransferRequest request) {
        return existing.getSenderId().equals(request.senderId())
                && existing.getRecipientId().equals(request.recipientId())
                && existing.getAmount().compareTo(request.amount()) == 0
                && existing.getCurrency().equals(request.currencyOrDefault());
    }

    private void publishNew(TransferEntity entity) {
        Transfer message = Transfer.newTransfer(entity.getId(), entity.getSenderId(), entity.getRecipientId(),
                entity.getAmount(), entity.getCurrency(), entity.getCreatedAt());
        try {
            template.send("transfers", message.id(), message).get(10, TimeUnit.SECONDS);
            LOG.info("Published {} for transfer {}", message.status(), message.id());
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            throw new ResponseStatusException(HttpStatus.SERVICE_UNAVAILABLE, KAFKA_UNAVAILABLE, e);
        } catch (ExecutionException | TimeoutException | KafkaException | org.apache.kafka.common.KafkaException e) {
            throw new ResponseStatusException(HttpStatus.SERVICE_UNAVAILABLE, KAFKA_UNAVAILABLE, e);
        }
    }
}

    private final TransferTrackingService trackingService;

        this.trackingService = trackingService;
    }

    @KafkaListener(id = "transfer-tracker",
            topics = {"transfers", "limit-results", "screening-results"},
            groupId = "transfer-tracker")
    public void onEvent(Transfer transfer, @Header(KafkaHeaders.RECEIVED_TOPIC) String topic) {
        LOG.info("Received {} for transfer {} from {}", transfer.status(), transfer.id(), topic);
        switch (topic) {
            case "limit-results" -> trackingService.recordServiceResult(transfer, HistoryStage.LIMIT);
            case "screening-results" -> trackingService.recordServiceResult(transfer, HistoryStage.SCREENING);
            case "transfers" -> {
                if (!Transfer.NEW.equals(transfer.status())) {
                    trackingService.applyFinalStatus(transfer);
                }
            }
            default -> LOG.warn("Unexpected topic {}", topic);
        }
    }

import org.springframework.stereotype.Service;
import com.example.common.Transfer;

@Service
public class TransferDecisionService {

    private static final String APPROVED = "APPROVED";
    private static final String DECLINED = "DECLINED";

    public Transfer decide(Transfer limitResult, Transfer screeningResult) {
        boolean limitApproved = APPROVED.equals(limitResult.status());
        boolean limitDeclined = DECLINED.equals(limitResult.status());
        boolean screeningApproved = APPROVED.equals(screeningResult.status());
        boolean screeningDeclined = DECLINED.equals(screeningResult.status());

        if (limitApproved && screeningApproved) {
            return limitResult.withDecision("COMPLETED", null, null);
        } else if (limitDeclined && screeningDeclined) {
            return limitResult.withDecision("DENIED", null,
                    limitResult.messageInfo() + "; " + screeningResult.messageInfo());
        } else if (limitDeclined && screeningApproved) {
            return limitResult.withDecision("CANCELLED", "limit", limitResult.messageInfo());
        } else if (limitApproved && screeningDeclined) {
            return limitResult.withDecision("CANCELLED", "screening", screeningResult.messageInfo());
        } else {
            return limitResult.withDecision("CANCELLED", null, "UNEXPECTED_RESULT: limit=" + limitResult.status()
                    + ", screening=" + screeningResult.status());
        }
    }
}

package com.example.transfer.service;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import com.example.common.Transfer;
import com.example.transfer.controller.TransferRequest;
import com.example.transfer.domain.HistoryStage;
import com.example.transfer.domain.TransferEntity;
import com.example.transfer.domain.TransferHistoryEntry;
import com.example.transfer.repository.TransferHistoryRepository;
import com.example.transfer.repository.TransferRepository;

import java.time.LocalDateTime;
import java.time.temporal.ChronoUnit;

@Service
public class TransferTrackingService {

    private static final Logger LOG = LoggerFactory.getLogger(TransferTrackingService.class);
    private final TransferRepository transferRepository;
    private final TransferHistoryRepository historyRepository;

    public TransferTrackingService(TransferRepository transferRepository,
                                   TransferHistoryRepository historyRepository) {
        this.transferRepository = transferRepository;
        this.historyRepository = historyRepository;
    }

    @Transactional
    public TransferEntity createTransfer(TransferRequest request) {
        LocalDateTime now = now();
        TransferEntity entity = new TransferEntity();
        entity.setId(request.id());
        entity.setSenderId(request.senderId());
        entity.setRecipientId(request.recipientId());
        entity.setAmount(request.amount());
        entity.setCurrency(request.currencyOrDefault());
        entity.setStatus("NEW");
        entity.setCreatedAt(now);
        entity.setUpdatedAt(now);
        transferRepository.saveAndFlush(entity);

        historyRepository.save(historyEntry(entity.getId(), HistoryStage.NEW, "NEW", null, null, now));

        LOG.info("Created transfer {} with status {}", entity.getId(), entity.getStatus());
        return entity;
    }

    @Transactional
    public void recordServiceResult(Transfer result, HistoryStage stage) {
        var entity = transferRepository.findById(result.id());
        if (entity.isEmpty()) {
            LOG.warn("Unknown transfer {} in {} result, skipped", result.id(), stage);
            return;
        }
        if (historyRepository.existsByTransferIdAndStage(result.id(), stage)) {
            LOG.info("Duplicate {} result for transfer {}, skipped", stage, result.id());
            return;
        }
        LocalDateTime now = now();
        historyRepository.save(historyEntry(
                result.id(), stage, result.status(), result.decidedBy(), result.messageInfo(), now));
        entity.get().setUpdatedAt(now);
        transferRepository.save(entity.get());
        LOG.info("Recorded {} result for transfer {}: {}", stage, result.id(), result.status());
    }

    @Transactional
    public void applyFinalStatus(Transfer result) {
        var found = transferRepository.findById(result.id());
        if (found.isEmpty()) {
            LOG.warn("Unknown transfer {} in final status, skipped", result.id());
            return;
        }
        if (historyRepository.existsByTransferIdAndStage(result.id(), HistoryStage.FINAL)) {
            LOG.info("Duplicate final status for transfer {}, skipped", result.id());
            return;
        }
        TransferEntity entity = found.get();
        entity.setStatus(result.status());
        entity.setDecidedBy(result.decidedBy());
        entity.setMessageInfo(result.messageInfo());
        LocalDateTime now = now();
        entity.setUpdatedAt(now);
        transferRepository.save(entity);

        historyRepository.save(historyEntry(
                entity.getId(), HistoryStage.FINAL, result.status(), result.decidedBy(), result.messageInfo(), now));

        LOG.info("Final status {} applied to transfer {}", entity.getStatus(), entity.getId());
    }

    private static LocalDateTime now() {
        return LocalDateTime.now().truncatedTo(ChronoUnit.MICROS);
    }

    private static TransferHistoryEntry historyEntry(Long transferId, HistoryStage stage, String status,
                                                     String decidedBy, String messageInfo, LocalDateTime changedAt) {
        TransferHistoryEntry entry = new TransferHistoryEntry(transferId, stage, status, decidedBy, messageInfo);
        entry.setChangedAt(changedAt);
        return entry;
    }
}
